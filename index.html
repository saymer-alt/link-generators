// === MIHOMO BUILDER (via web4core) === function buildMihomo() { const raw = document.getElementById('mihomoInput').value.trim(); if (!raw && wgBeans.length === 0) return showToast('Вставь ссылки или загрузи WG конфиг', true); const api = globalThis.web4core; if (!api?.buildFromRequest) return showToast('❌ web4core.runtime не загружен!', true); const urlTest = document.getElementById('pingSelect').value || 'https://google.com/generate_204'; const addTun = document.getElementById('cfgTun').checked; const perProxyTun = document.getElementById('cfgPerProxyTun').checked; const perProxySocks = document.getElementById('cfgPerProxySocks').checked; const useChain = document.getElementById('cfgChain').checked; try { const result = api.buildFromRequest({ core: 'mihomo', input: raw, wgBeans: wgBeans, options: { addSocks: document.getElementById('cfgSocks').checked, addTun: addTun, webUI: document.getElementById('cfgWebUI').checked, urlTest: urlTest, mihomoSubscriptionMode: document.getElementById('cfgSubMode').checked, mihomoPerProxyTun: perProxyTun, perProxyPort: perProxySocks } }); let yamlRaw = typeof result.data === 'string' ? result.data : JSON.stringify(result.data, null, 2); // PATCH 1: Allow LAN (через текстовую замену, чтобы не ломать шаблон) if (document.getElementById('cfgLan').checked) { yamlRaw = yamlRaw.replace(/^allow-lan:\s*false\s*$/m, 'allow-lan: true'); if (!yamlRaw.includes('bind-address')) { yamlRaw = yamlRaw.replace(/^(allow-lan:\s*true\s*)$/m, '$1\nbind-address: "*"'); } } // PATCH 2: WARP-in-WARP (Цепочки) через безопасную текстовую инъекцию if (useChain) { try { // Парсим YAML в объект ТОЛЬКО для того, чтобы узнать имена прокси const configObj = jsyaml.load(yamlRaw); if (configObj && configObj.proxies && configObj.proxies.length > 0) { const outerWg = configObj.proxies.find(p => p.type === 'wireguard'); if (outerWg) { let chainedCount = 0; configObj.proxies.forEach(p => { if (p.type === 'masque') { // Экранируем спецсимволы в имени для регулярного выражения const escapeRegExp = (string) => string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); const nameRegex = new RegExp(`(name:\\s*["']?${escapeRegExp(p.name)}["']?)`, 'g'); // Вставляем dialer-proxy сразу под именем нужного MASQUE-туннеля yamlRaw = yamlRaw.replace(nameRegex, `$1\n dialer-proxy: ${outerWg.name}`); chainedCount++; } }); if (chainedCount === 0) showToast('⚠️ MASQUE туннели для цепочки не найдены', true); } else { showToast('⚠️ Загрузите WG конфиг для создания цепочки', true); } } } catch(e) { console.error("Ошибка при сборке цепочки:", e); } } document.getElementById('mihomoOutput').value = yamlRaw; showToast(useChain ? '✅ WARP-in-WARP конфиг собран!' : '✅ Mihomo config built!'); } catch (e) { showToast('❌ ' + (e.message || e), true); } }
