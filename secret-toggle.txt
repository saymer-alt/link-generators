    if (f.secret) { inp.placeholder = CS_SECRET_INPUT_HINT; inp.dataset.csSecret = '1'; }
    else inp.value = String(f.value);
    row.append(lbl, inp);
    if (f.secret) {
      // TRACK C: явный показать/скрыть ТЕКУЩЕГО значения (владелец вправе его
      // видеть и править). Показ = подстановка оригинала в input; скрытие =
      // возврат к семантике «пусто = оставить исходное». В диагностику/логи
      // значение по-прежнему не попадает.
      const reveal = document.createElement('button');
      reveal.type = 'button';
      reveal.className = 'btn btn-gray';
      reveal.textContent = '👁';
      reveal.setAttribute('aria-label', 'Показать или скрыть значение поля ' + f.key + ' объекта ' + cur);
      reveal.title = 'Показать/скрыть текущее значение';
      let shown = false;
      reveal.addEventListener('click', () => {
        shown = !shown;
        inp.type = shown ? 'text' : 'password';
        inp.value = shown ? String(f.value) : '';
        inp.placeholder = shown ? '' : CS_SECRET_INPUT_HINT;
        reveal.textContent = shown ? '🙈' : '👁';
      });
      row.append(reveal);
    }
    form.appendChild(row);
